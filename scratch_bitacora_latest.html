function abrirBitacora(id, sn) { $('#b-equipo-id').val(id); // Inicializar select2 dentro del modal para evitar problemas de z-index $('#modalBitacora .select2').select2({ dropdownParent: $('#modalBitacora'), width: '100%' }); // Fetch equipo details for the header card $.ajax({ url: '/equipos/api/' + id + '/ver/', type: 'GET', success: function(resp) { var eq = resp.data; $('#b-imagen').attr('src', eq.imagen || '/static/img/placeholder_equipo.png'); $('#b-equipo-sn-header').text(eq.serial_number || sn); $('#b-equipo-estado').text(eq.estado || 'S/E'); $('#b-equipo-ubicacion').text((eq.edificio || '') + ' - ' + (eq.unidad || '')); // Preseleccionar unidad if (eq.unidad) { $('#b-unidad').val(eq.unidad).trigger('change'); } } }); $('#form-bitacora')[0].reset(); $('#b-tipo').val('').trigger('change'); $('#b-falla').val('').trigger('change');
